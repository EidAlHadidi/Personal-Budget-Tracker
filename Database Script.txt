create procedure [dbo].[sp_DisplayGoalsForUser]
@UserID int
as
begin
	select SavingGoalID,Categories.CategoryName,GoalName,Description,TargetAmount,CurrentAmount,StartDate,EndDate,IsCompleted
	from SavingGoals
	inner join Categories on SavingGoals.CategoryID = Categories.CategoryID
	where UserID = @UserID
end
GO

CREATE procedure [dbo].[sp_DisplayTransactionsForUser]
@UserID int
as
begin
	select 
	TransactionID,
	TransactionTypes.TransactionTypeName,
	Categories.CategoryName,
	Date,
	Time,
	Description,
	Amount,
	GoalID
	from transactions 
	inner join TransactionTypes on Transactions.TransactionTypeID = TransactionTypes.TransactionTypeID
	left outer join Categories on Categories.CategoryID = Transactions.CategoryID
	where UserID = @UserID
end
GO

CREATE procedure [dbo].[sp_GetTotalMoneySaved]
@CurrentSavedMoney decimal(19,4) output,
@TotalSavedMoney decimal(19,4) output,
@UserID int
as
begin
	select @CurrentSavedMoney = sum(CurrentAmount) from 
	SavingGoals where UserID = @UserID and IsCompleted = 0;
	select @TotalSavedMoney = sum(CurrentAmount) from
	SavingGoals where UserID = @UserID;

	if @CurrentSavedMoney is null 
		set @CurrentSavedMoney = 0;
	if @TotalSavedMoney is null
		set @TotalSavedMoney = 0;

end
